## 基础示例

### 示例 1: 单任务执行

```python
import asyncio
from shared.subagents.dispatcher import create_dispatcher, create_task

async def single_task_example():
    """最简单的单任务执行示例"""

    # 创建任务
    task = create_task(
        task_id="hello-world",
        name="Hello World",
        description="打印 Hello World",
        requirements="输出 'Hello, World!'"
    )

    # 创建调度器
    dispatcher = create_dispatcher(mode="sequential")

    # 执行任务
    results = await dispatcher.execute([task])

    # 查看结果
    result = results[0]
    print(f"任务状态: {result.status.value}")
    print(f"任务输出: {result.output}")
    print(f"执行时间: {result.execution_time:.2f}秒")

asyncio.run(single_task_example())
```

**输出**：
```
任务状态: completed
任务输出: 任务 Hello World 已完成
执行时间: 0.15秒
```

### 示例 2: 多任务顺序执行

```python
async def sequential_tasks_example():
    """多个任务按顺序执行"""

    tasks = [
        create_task(
            task_id="task-1",
            name="创建项目结构",
            description="创建目录和文件",
            requirements="""
            1. 创建 src/ 目录
            2. 创建 tests/ 目录
            3. 创建 README.md
            """,
            priority=1
        ),
        create_task(
            task_id="task-2",
            name="编写核心代码",
            description="实现主要功能",
            requirements="""
            1. 实现 main() 函数
            2. 添加错误处理
            """,
            priority=2,
            dependencies=["task-1"]  # 依赖 task-1
        ),
        create_task(
            task_id="task-3",
            name="编写测试",
            description="单元测试",
            requirements="测试 main() 函数",
            priority=3,
            dependencies=["task-2"]  # 依赖 task-2
        ),
    ]

    dispatcher = create_dispatcher(mode="sequential")
    results = await dispatcher.execute(tasks)

    # 汇总结果
    for result in results:
        print(f"✅ {result.task_id}: {result.status.value}")

asyncio.run(sequential_tasks_example())
```

### 示例 5: 自定义代码审查规则

```python
from shared.subagents.reviewer import (
    ReviewRule, CodeReviewIssue,
    IssueCategory, Severity, create_reviewer
)

class PerformanceRule(ReviewRule):
    """自定义性能规则"""

    def __init__(self):
        super().__init__(
            name="performance",
            category=IssueCategory.PERFORMANCE,
            default_severity=Severity.IMPORTANT
        )

    def check(self, code, file_path=""):
        issues = []

        # 检查 N+1 查询模式
        if "for" in code and "SELECT" in code:
            issues.append(CodeReviewIssue(
                severity=Severity.IMPORTANT,
                category=self.category.value,
                description="可能的 N+1 查询问题",
                location=file_path,
                suggestion="使用 JOIN 或预加载优化"
            ))

        return issues

async def custom_review_example():
    """使用自定义审查规则"""

    # 创建带自定义规则的审查器
    reviewer = create_reviewer(
        custom_rules=[PerformanceRule()]
    )

    # 使用审查器
    from shared.subagents.dispatcher import Dispatcher
    dispatcher = Dispatcher(reviewer=reviewer)

    # 执行任务（会使用自定义规则审查）
    # ...

asyncio.run(custom_review_example())
```

            task_type="testing",
            dependencies=["auth-api"]

## 技巧和最佳实践

### 1. 任务命名规范

```python
# ✅ 好的命名
create_task(
    task_id="user-auth-api",
    name="实现用户认证 API",
    requirements="..."
)

# ❌ 不好的命名
create_task(
    task_id="task-1",
    name="做点什么",
    requirements="..."
)
```

### 2. 依赖管理

```python
# ✅ 明确依赖
create_task(
    task_id="test",
    dependencies=["api", "model"]  # 依赖多个任务
)

# ❌ 隐式依赖
create_task(
    task_id="test",
    # 没有声明依赖，假设顺序执行
)
```

### 3. 复杂度评估

```python
# ✅ 合理的复杂度
create_task(
    name="实现算法",
    complexity="high",  # 高复杂度 → 使用 Opus
    task_type="development"
)

# ❌ 低估复杂度
create_task(
    name="实现机器学习模型",
    complexity="low",  # 应该是 high
)
```

### 4. 错误处理

```python
async def robust_execution():
    """健壮的任务执行"""

    try:
        workflow = create_workflow()
        results = await workflow.execute(tasks)

        # 检查失败的任务
        failed = [r for r in results if not r.success]

        if failed:
            print(f"⚠️  {len(failed)} 个任务失败")

            # 重试失败的 tasks
            retry_tasks = [t for t in tasks if t.id in [f.task_id for f in failed]]
            retry_results = await workflow.execute(retry_tasks)

        return results

    except Exception as e:
        print(f"❌ 执行失败: {e}")
        return []
```
